# nixthepouch.com

# Conventional crawlers — everything is public, nothing to hide.
# The one exception is the post template: scaffolding with placeholder copy, not a page.
# Jekyll on GitHub Pages normally refuses to publish underscore-prefixed files, so the
# Disallow is a second lock rather than the only one.
User-agent: *
Allow: /
Disallow: /blog/_template.html

# AI assistants and answer engines are allowed deliberately, not by default.
# Being cited by ChatGPT, Claude, Perplexity and Google AI Overviews is a primary
# acquisition channel for this site — a person asking "how do I quit Zyn" is exactly
# the reader NIX is for. Blocking these would trade away that channel to protect
# marketing copy that is already public.
